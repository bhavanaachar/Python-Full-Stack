
##########################################################################
7. str (STRING)
##########################################################################
DEFINITION
  A string is a sequence of characters (letters, digits, symbols, spaces)
  enclosed in quotes. Strings are immutable.

CREATING STRINGS
  s1 = 'Hello'                # single quotes
  s2 = "World"                # double quotes
  s3 = """This is a
  multi-line string"""        # triple quotes
  empty = ""
  print(type(s1))             # <class 'str'>

QUOTES INSIDE STRINGS
  print("It's a nice day")
  print('He said "Hi"')
  print("He said \"Hi\"")     # escape character
  print('It\'s fine')

ESCAPE CHARACTERS
  \n   new line          print("Hello\nWorld")
  \t   tab               print("A\tB")
  \\   backslash         print("C:\\folder")
  \'   single quote
  \"   double quote

RAW STRING (no escapes)
  print(r"C:\new\table")      # C:\new\table

INDEXING
  s = "PYTHON"
  Index:    0  1  2  3  4  5
  Char:     P  Y  T  H  O  N
  Negative: -6 -5 -4 -3 -2 -1

  print(s[0])                 # P
  print(s[3])                 # H
  print(s[-1])                # N

SLICING   string[start : stop : step]   (stop is NOT included)
  print(s[0:3])               # PYT
  print(s[2:])                # THON
  print(s[:4])                # PYTH
  print(s[::2])               # PTO
  print(s[::-1])              # NOHTYP   (reverse)

IMMUTABLE
  s = "Python"
  # s[0] = "J"                # TypeError
  s = "J" + s[1:]             # create a new string
  print(s)                    # Jython

OPERATIONS
  print("Hello" + " " + "World")   # Hello World   (concatenation)
  print("Hi " * 3)                 # Hi Hi Hi      (repetition)
  print(len("Python"))             # 6
  print("Py" in "Python")          # True
  print("abc" < "abd")             # True  (alphabetical)

COMMON METHODS
  s = "  Hello World  "
  print(s.upper())            #   HELLO WORLD
  print(s.lower())            #   hello world
  print(s.strip())            # Hello World
  print(s.title())            #   Hello World
  print(s.replace("World", "Python"))
  print("a,b,c".split(","))   # ['a', 'b', 'c']
  print("-".join(["a", "b"])) # a-b
  print("hello".find("l"))    # 2
  print("banana".count("a"))  # 3
  print("hello".startswith("he"))   # True
  print("123".isdigit())      # True
  print("abc".isalpha())      # True

FORMATTING
  name = "Ravi"
  age = 20
  print(f"{name} is {age} years old")           # f-string
  print("{} is {} years old".format(name, age))
  print("%s is %d years old" % (name, age))
  print(f"Pi = {3.14159:.2f}")                   # Pi = 3.14

LOOPING THROUGH A STRING
  for ch in "Hi!":
      print(ch)

MULTI-LINE STRING / DOCSTRING
  text = """Line 1
  Line 2
  Line 3"""
  print(text)

POINTS TO REMEMBER
  - Index starts from 0; out-of-range index gives IndexError.
  - Slicing never gives an error for out-of-range values.
  - Methods return NEW strings; the original is unchanged.
  - input() always returns a str.
